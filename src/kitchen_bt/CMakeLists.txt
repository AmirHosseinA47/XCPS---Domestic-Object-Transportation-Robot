cmake_minimum_required(VERSION 3.16)
project(kitchen_bt)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(kitchen_interfaces REQUIRED)
find_package(ament_index_cpp REQUIRED)

add_executable(kitchen_bt_runner src/kitchen_bt_runner.cpp)
target_link_libraries(kitchen_bt_runner behaviortree_cpp::behaviortree_cpp)
ament_target_dependencies(kitchen_bt_runner rclcpp kitchen_interfaces ament_index_cpp)
install(TARGETS kitchen_bt_runner DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY behavior_trees DESTINATION share/${PROJECT_NAME})
ament_package()
